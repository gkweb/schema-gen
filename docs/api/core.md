# @schema-gen/core

The core package provides the main generator functionality, configuration loading, and native Rust bindings.

## Installation

```bash
pnpm add @schema-gen/core
```

## Configuration

### defineConfig

Type-safe configuration helper.

```ts
import { defineConfig } from '@schema-gen/core';

export default defineConfig({
  input: { path: './openapi.yaml' },
  output: { dir: './src/api' },
  plugins: ['typescript-types'],
});
```

### loadConfig

Load configuration from the current directory.

```ts
import { loadConfig } from '@schema-gen/core';

const result = await loadConfig();
// result: { config: UserConfig, filepath: string } | null
```

### loadConfigFromFile

Load configuration from a specific file.

```ts
import { loadConfigFromFile } from '@schema-gen/core';

const result = await loadConfigFromFile('./my-config.ts');
// result: { config: UserConfig, filepath: string }
```

### validateConfig

Validate a configuration object.

```ts
import { validateConfig } from '@schema-gen/core';

const errors = validateConfig(config);
// errors: string[] (empty if valid)
```

### createDefaultConfig

Create a default configuration.

```ts
import { createDefaultConfig } from '@schema-gen/core';

const config = createDefaultConfig();
```

## Generator

### createGenerator

Create a generator instance.

```ts
import { createGenerator } from '@schema-gen/core';

const generator = await createGenerator({
  config: myConfig,
  baseDir: process.cwd(),
});

// Generate files (returns GeneratedFile[])
const files = await generator.run();

// Write files to disk
await generator.write(files);

// Or do both in one step
await generator.generate();
```

## Spec Utilities

### parseSpec

Parse an OpenAPI specification to AST.

```ts
import { parseSpec } from '@schema-gen/core';

const ast = parseSpec(specContent);
// ast: SchemaAst
```

### parseSpecToJson

Parse an OpenAPI specification and return JSON string.

```ts
import { parseSpecToJson } from '@schema-gen/core';

const json = parseSpecToJson(specContent);
// json: string (JSON representation of AST)
```

### validateSpec

Validate an OpenAPI specification.

```ts
import { validateSpec } from '@schema-gen/core';

validateSpec(specContent);
// Throws if invalid
```

## Version

### getVersion

Get the schema-gen version.

```ts
import { getVersion } from '@schema-gen/core';

const version = getVersion();
// version: string (e.g., '0.2.0')
```

## Types

### UserConfig

User-facing configuration type.

```ts
interface UserConfig {
  cwd?: string;
  input: {
    path: string;
    validation?: 'strict' | 'warn' | 'off';
    parserOptions?: ParserOptions;
  };
  output: {
    dir: string;
    structure?: 'flat' | 'by-tag' | 'by-endpoint';
    clean?: boolean;
    types?: TypesOutputConfig;
  };
  plugins: PluginConfig[];
  transform?: TransformConfig;
  fetch?: FetchConfig;
  reactQuery?: ReactQueryConfig;
  vueQuery?: VueQueryConfig;
  style?: StyleConfig;
}
```

### SchemaGenConfig

Resolved configuration type (alias for UserConfig).

```ts
type SchemaGenConfig = UserConfig;
```

### GeneratedFile

A file generated by a plugin.

```ts
interface GeneratedFile {
  path: string;
  content: string;
}
```

### PluginConfig

Plugin configuration format.

```ts
type PluginConfig =
  | string                              // Built-in name or npm package
  | { name: string; config?: object }   // With configuration
  | PluginObject;                       // Direct plugin object
```

### ParserOptions

Options for parsing remote schemas.

```ts
interface ParserOptions {
  resolve?: {
    http?: {
      headers?: Record<string, string>;
      timeout?: number;
    };
  };
}
```

### TransformConfig

Global transform options.

```ts
interface TransformConfig {
  naming?: {
    types?: 'PascalCase' | 'camelCase' | 'preserve';
    properties?: 'PascalCase' | 'camelCase' | 'snake_case' | 'preserve';
    enums?: 'PascalCase' | 'SCREAMING_SNAKE' | 'preserve';
  };
  typeOverrides?: Record<string, string>;
  include?: { tags?: string[]; paths?: string[] };
  exclude?: { tags?: string[]; operationIds?: string[] };
}
```

### StyleConfig

Code style options.

```ts
interface StyleConfig {
  semi?: boolean;
  quotes?: 'single' | 'double';
  trailingComma?: 'all' | 'es5' | 'none';
  tabWidth?: number;
  useTabs?: boolean;
  printWidth?: number;
}
```
